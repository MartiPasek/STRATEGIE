# PO (Phase HA-1 — 2 upstreams):

strategie-ai.com, www.strategie-ai.com, app.strategie-ai.com, api.strategie-ai.com {
    # ... existing tls + headers + matchers ...

    # Phase HA-1 (23.5.2026): zero-downtime restart pres 2 API upstreams.
    # Caddy active health check /api/v1/health (no auth, lightweight).
    # Pokud instance returns non-200 (down for restart, crash), Caddy
    # automaticky failover na zbyvajici upstream do dalsi pasive check.
    reverse_proxy localhost:8001 localhost:8002 {
        # Load balance — least_conn = preferuje upstream s méně active connections
        # (round_robin alternative = rovnoměrné distribution)
        lb_policy least_conn

        # Active health check — Caddy polling each upstream
        health_uri /api/v1/health
        health_interval 3s
        health_timeout 2s
        health_status 2xx
        # Po fail: instance removed z pool, retry za N*interval
        fail_duration 10s
        max_fails 1

        # Header pass-through (existing behavior)
        header_up Host {host}
        header_up X-Real-IP {remote_host}
        header_up X-Forwarded-For {remote_host}
        header_up X-Forwarded-Proto {scheme}
    }
}
