# Najdi blok pro strategie-ai.com v C:\caddy\Caddyfile a nahrad
# `reverse_proxy localhost:8002` (nebo 8001) za:

reverse_proxy localhost:8002 localhost:8003 {
    # Load balance - least_conn = preferuj idle upstream
    lb_policy least_conn

    # Active health check
    health_uri /api/v1/health
    health_interval 3s
    health_timeout 2s
    health_status 2xx

    # Failover - dead upstream removed z pool na 10s
    fail_duration 10s
    max_fails 1

    # Existing headers
    header_up Host {host}
    header_up X-Real-IP {remote_host}
    header_up X-Forwarded-For {remote_host}
    header_up X-Forwarded-Proto {scheme}
}
